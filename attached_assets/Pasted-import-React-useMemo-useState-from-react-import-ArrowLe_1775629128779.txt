function PhoneCodeScreen({ code, setCode, onBack, onNext }) {
  const digits = useMemo(() => {
    const arr = ["", "", "", ""];
    code.slice(0, 4).split("").forEach((digit, i) => {
      arr[i] = digit;
    });
    return arr;
  }, [code]);

  const updateDigit = (index, value) => {
    const clean = value.replace(/\D/g, "").slice(-1);
    const next = [...digits];
    next[index] = clean;
    setCode(next.join(""));

    if (clean) {
      const nextEl = document.querySelector(`[data-otp-phone='${index + 1}']`);
      if (nextEl) nextEl.focus();
    }
  };

  return (
    <div className={screenFrame}>
      <StatusBar />

      <div className="px-4 pt-20">
        <div className="max-w-[280px] text-[22px] font-semibold leading-7 tracking-[-0.03em] text-[#111]">
          Enter the 4-digit code sent to you at 05599224612
        </div>

        <div className="mt-8 flex gap-2">
          {digits.map((digit, index) => (
            <input
              key={index}
              data-otp-phone={index}
              value={digit}
              onChange={(e) => updateDigit(index, e.target.value)}
              className={otpInputClass}
              inputMode="numeric"
              maxLength={1}
            />
          ))}
        </div>

        <button type="button" className="mt-6 rounded-full bg-[#f1f1f1] px-4 py-2 text-[12px] text-[#8c8c8c]">
          I haven’t received a code (0:09)
        </button>

        <button type="button" className="mt-4 block rounded-full bg-[#f1f1f1] px-4 py-2 text-[13px] font-medium text-[#111]">
          Log in with password
        </button>
      </div>

      <FooterNav onBack={onBack} onNext={onNext} nextDisabled={code.length !== 4} />
      <BottomHomeBar />
    </div>
  );
}
